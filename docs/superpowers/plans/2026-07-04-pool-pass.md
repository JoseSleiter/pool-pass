# Pool & Services Pass PWA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A mobile-first PWA where customers book pool day passes, pay via Mercantil C2P (Pago Móvil) in VES at the BCV rate, and receive a QR voucher by email that gate staff scan to check them in.

**Architecture:** Single Next.js App Router app on Vercel. Server Actions hold all payment/booking logic; a Postgres function `reserve_slots()` in Supabase does the atomic capacity check (advisory lock per date). Supabase Auth protects `/admin` only — customers are guests. Resend sends the voucher email with a QR image served from `/api/qr/[id]`.

**Tech Stack:** Next.js (App Router, TypeScript, `src/` dir), Tailwind CSS + shadcn/ui, Supabase (Postgres + Auth via `@supabase/ssr`), zod + react-hook-form, Resend + react-email, `qrcode`, Vitest + Testing Library (unit), Playwright (e2e).

**Spec:** `docs/superpowers/specs/2026-07-04-pool-pass-design.md`

**Conventions:**
- Customer-facing UI copy is in Spanish (Venezuelan audience); admin UI and code are in English.
- Dates are `YYYY-MM-DD` strings end to end.
- Money: USD and VES as `number`, rounded to 2 decimals with `roundMoney()` — never floating math without it.
- All secrets only in server code. The client never computes authoritative prices.
- Integration tests (marked `*.integration.test.ts`) need the local Supabase stack running (`supabase start`). Unit tests never touch the network or DB.

```sql
-- Settings: single row
create table settings (
  id int primary key default 1 check (id = 1),
  daily_capacity int not null default 100,
  adult_price_usd numeric(10,2) not null default 15,
  child_price_usd numeric(10,2) not null default 5,
  last_bcv_rate numeric(14,4),
  last_bcv_rate_at timestamptz
);
insert into settings (id) values (1);

-- Per-date capacity override; 0 = closed
create table capacity_overrides (
  date date primary key,
  capacity int not null check (capacity >= 0)
);

create table reservations (
  id uuid primary key default gen_random_uuid(),
  code text not null unique,
  date date not null,
  first_name text not null,
  last_name text not null,
  email text not null,
  id_type text not null check (id_type in ('V','E','P','J')),
  id_number text not null,
  adults int not null check (adults >= 1),
  children int not null check (children >= 0),
  total_usd numeric(10,2) not null,
  bcv_rate numeric(14,4),
  total_ves numeric(14,2),
  status text not null default 'pending_payment'
    check (status in ('pending_payment','confirmed','failed','expired','checked_in')),
  payment_ref text,
  expires_at timestamptz,
  checked_in_at timestamptz,
  created_at timestamptz not null default now()
);
create index reservations_date_idx on reservations (date);

-- RLS: nothing for anon; read/update for authenticated admins.
-- The app server uses the service role, which bypasses RLS.
alter table settings enable row level security;
alter table capacity_overrides enable row level security;
alter table reservations enable row level security;

create policy "admin read settings" on settings for select to authenticated using (true);
create policy "admin update settings" on settings for update to authenticated using (true);
create policy "admin all overrides" on capacity_overrides for all to authenticated using (true);
create policy "admin read reservations" on reservations for select to authenticated using (true);
create policy "admin update reservations" on reservations for update to authenticated using (true);

-- Atomic capacity check + pending insert (see spec: capacity counting rule)
create or replace function reserve_slots(
  p_date date,
  p_first_name text,
  p_last_name text,
  p_email text,
  p_id_type text,
  p_id_number text,
  p_adults int,
  p_children int,
  p_total_usd numeric
) returns table (ok boolean, reservation_id uuid, code text, remaining int)
language plpgsql
as $$
declare
  v_limit int;
  v_occupancy int;
  v_code text;
begin
  -- serialize all bookings for the same date
  perform pg_advisory_xact_lock(hashtext('pool-' || p_date::text));

  select coalesce(
    (select o.capacity from capacity_overrides o where o.date = p_date),
    (select s.daily_capacity from settings s where s.id = 1)
  ) into v_limit;

  select coalesce(sum(r.adults + r.children), 0) into v_occupancy
  from reservations r
  where r.date = p_date
    and (r.status in ('confirmed','checked_in')
         or (r.status = 'pending_payment' and r.expires_at > now()));

  if v_occupancy + p_adults + p_children > v_limit then
    return query select false, null::uuid, null::text, greatest(v_limit - v_occupancy, 0);
    return;
  end if;

  v_code := 'PP-' || upper(substring(md5(gen_random_uuid()::text) from 1 for 5));

  return query
  insert into reservations
    (code, date, first_name, last_name, email, id_type, id_number,
     adults, children, total_usd, status, expires_at)
  values
    (v_code, p_date, p_first_name, p_last_name, p_email, p_id_type, p_id_number,
     p_adults, p_children, p_total_usd, 'pending_payment', now() + interval '10 minutes')
  returning true, reservations.id, reservations.code, null::int;
end;
$$;
```

```ts
import { describe, it, expect, beforeEach } from 'vitest';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

const DATE = '2030-01-15';

function reserve(adults: number, children = 0) {
  return supabase.rpc('reserve_slots', {
    p_date: DATE,
    p_first_name: 'Ana',
    p_last_name: 'Pérez',
    p_email: 'ana@example.com',
    p_id_type: 'V',
    p_id_number: '12345678',
    p_adults: adults,
    p_children: children,
    p_total_usd: adults * 15 + children * 5,
  });
}

describe('reserve_slots', () => {
  beforeEach(async () => {
    await supabase.from('reservations').delete().eq('date', DATE);
    await supabase.from('capacity_overrides').delete().eq('date', DATE);
    await supabase.from('capacity_overrides').insert({ date: DATE, capacity: 10 });
  });

  it('accepts a booking within capacity and returns a code', async () => {
    const { data, error } = await reserve(2, 1);
    expect(error).toBeNull();
    const row = data![0];
    expect(row.ok).toBe(true);
    expect(row.code).toMatch(/^PP-[A-Z0-9]{5}$/);
  });

  it('rejects when the request exceeds remaining capacity, reporting remaining', async () => {
    await reserve(8); // 8 of 10 taken
    const { data } = await reserve(3);
    const row = data![0];
    expect(row.ok).toBe(false);
    expect(row.remaining).toBe(2);
  });

  it('treats capacity 0 override as closed', async () => {
    await supabase.from('capacity_overrides').update({ capacity: 0 }).eq('date', DATE);
    const { data } = await reserve(1);
    expect(data![0].ok).toBe(false);
    expect(data![0].remaining).toBe(0);
  });

  it('never oversells under concurrent requests', async () => {
    // 6 concurrent requests of 3 people each against capacity 10 → at most 3 can win
    const results = await Promise.all(Array.from({ length: 6 }, () => reserve(3)));
    const accepted = results.filter((r) => r.data![0].ok).length;
    expect(accepted).toBe(3);
    const { data: rows } = await supabase
      .from('reservations')
      .select('adults, children')
      .eq('date', DATE);
    const total = rows!.reduce((s, r) => s + r.adults + r.children, 0);
    expect(total).toBeLessThanOrEqual(10);
  });
});
```

```ts
export function roundMoney(n: number): number {
  return Math.round(n * 100) / 100;
}

export function totalUsd(
  adults: number,
  children: number,
  prices: { adult: number; child: number }
): number {
  return roundMoney(adults * prices.adult + children * prices.child);
}

export function usdToVes(usd: number, rate: number): number {
  return roundMoney(usd * rate);
```

Create `src/lib/bcv.ts`:

```ts
const FRESH_MS = 60 * 60 * 1000; // serve from cache for 1h
const MAX_STALE_MS = 24 * 60 * 60 * 1000; // fallback window on fetch failure

export interface RateDeps {
  fetchRate: () => Promise<number>;
  readCache: () => Promise<{ rate: number; fetchedAt: number } | null>;
  writeCache: (rate: number, fetchedAt: number) => Promise<void>;
  now: () => number;
}

/** Returns the BCV VES/USD rate, or null if no acceptable rate is available. */
export async function getBcvRate(deps: RateDeps): Promise<number | null> {
  const cached = await deps.readCache();
  const now = deps.now();

  if (cached && now - cached.fetchedAt < FRESH_MS) return cached.rate;

  try {
    const rate = await deps.fetchRate();
    await deps.writeCache(rate, now);
    return rate;
  } catch {
    if (cached && now - cached.fetchedAt < MAX_STALE_MS) return cached.rate;
    return null;
  }
}

/** Production fetcher: dolarapi "oficial" mirror of the BCV rate. */
export async function fetchBcvRateFromMirror(): Promise<number> {
  const res = await fetch(process.env.BCV_RATE_URL!, { cache: 'no-store' });
  if (!res.ok) throw new Error(`BCV mirror HTTP ${res.status}`);
  const body = (await res.json()) as { promedio: number };
  if (typeof body.promedio !== 'number' || body.promedio <= 0) {
    throw new Error('BCV mirror returned invalid rate');
  }
  return body.promedio;
}
```

```ts
const SECRET = 'test-secret-key';

function decrypt(b64: string): string {
  const key = createHash('sha256').update(SECRET).digest();
  const iv = Buffer.alloc(16, 0);
```

```ts
import { createCipheriv, createHash } from 'crypto';

/**
 * AES-256-CBC, key = SHA-256(secret), zero IV, base64 output.
 * Cipher convention per spec — verify against Mercantil's integration
 * doc before the production smoke test. Only this file changes if it differs.
 */
export function encryptField(value: string, secret: string): string {
  const key = createHash('sha256').update(secret).digest();
  const iv = Buffer.alloc(16, 0);
  const cipher = createCipheriv('aes-256-cbc', key, iv);
  return Buffer.concat([cipher.update(value, 'utf8'), cipher.final()]).toString('base64');
}
```

```ts
export interface C2pRequest {

  mock: boolean;

    clientId: process.env.MERCANTIL_CLIENT_ID!,
    merchantId: process.env.MERCANTIL_MERCHANT_ID!,

    terminalId: process.env.MERCANTIL_TERMINAL_ID ?? '1',
    secretKey: process.env.MERCANTIL_SECRET_KEY!,
    mock: process.env.MERCANTIL_MOCK === '1',
  };

export async function payC2p(req: C2pRequest, cfg: MercantilConfig): Promise<C2pResult> {
  if (cfg.mock) {
    return req.otp === '999999'
      ? { status: 'declined', message: 'Clave inválida (mock)' }
      : { status: 'approved', reference: `MOCK-${Date.now()}` };
  }

  const payload = {
    merchant_identify: {
      integratorId: Number(cfg.integratorId),
      merchantId: cfg.merchantId,
      terminalId: cfg.terminalId,
    },
    client_identify: { ipaddress: '127.0.0.1', browser_agent: 'PoolPass/1.0' },
    transaction_c2p: {
      trx_type: 'compra',
      payment_method: 'c2p',
      currency: 'ves',
      amount: req.amountVes,
      destination_bank_id: Number(req.destinationBankCode),
      destination_id: encryptField(req.destinationId, cfg.secretKey),
      destination_mobile_number: encryptField(req.destinationPhone, cfg.secretKey),
      twofactor_auth: encryptField(req.otp, cfg.secretKey),
      invoice_number: req.invoiceNumber,
    },
  };

  try {
    const res = await fetch(`${cfg.baseUrl}/v1/payment/pay`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-IBM-Client-Id': cfg.clientId },
      body: JSON.stringify(payload),
    });
    const body = await res.json().catch(() => null);
    const trx = body?.transaction_c2p_response;
    if (trx?.trx_status === 'approved') {
      return { status: 'approved', reference: String(trx.payment_reference ?? '') };
    }
    if (trx?.trx_status) {
      return { status: 'declined', message: String(trx.trx_status_description ?? 'Pago rechazado') };
    }
    return { status: 'unknown', message: `Respuesta ambigua del banco (HTTP ${res.status})` };
  } catch (e) {
    return { status: 'unknown', message: e instanceof Error ? e.message : 'Fallo de red' };
  }
```

```ts
  const s = await getSettings();
  if (s.last_bcv_rate == null || s.last_bcv_rate_at == null) return null;
  return { rate: Number(s.last_bcv_rate), fetchedAt: Date.parse(s.last_bcv_rate_at) };
}

export async function writeRateCache(rate: number, fetchedAt: number): Promise<void> {
  const { error } = await serviceClient()
    .from('settings')
    .update({ last_bcv_rate: rate, last_bcv_rate_at: new Date(fetchedAt).toISOString() })
    .eq('id', 1);
  if (error) throw new Error(error.message);
}

/** Remaining capacity per day for a month ('YYYY-MM'). Missing key = default capacity untouched. */
export async function monthAvailability(month: string): Promise<Record<string, number>> {
  const db = serviceClient();
  const first = `${month}-01`;
  const last = new Date(Date.UTC(
    Number(month.slice(0, 4)), Number(month.slice(5, 7)), 0
  )).toISOString().slice(0, 10);

  const [settings, overridesRes, rowsRes] = await Promise.all([
    getSettings(),
    db.from('capacity_overrides').select('*').gte('date', first).lte('date', last),
    db.from('reservations')
      .select('date, adults, children, status, expires_at')
      .gte('date', first).lte('date', last)
      .in('status', ['pending_payment', 'confirmed', 'checked_in']),
  ]);
  if (overridesRes.error) throw new Error(overridesRes.error.message);
  if (rowsRes.error) throw new Error(rowsRes.error.message);

  const now = Date.now();
  const occupancy: Record<string, number> = {};
  for (const r of rowsRes.data) {
    const counts =
      r.status !== 'pending_payment' ||
      (r.expires_at !== null && Date.parse(r.expires_at) > now);
    if (counts) occupancy[r.date] = (occupancy[r.date] ?? 0) + r.adults + r.children;
  }

  const result: Record<string, number> = {};
  const overrides = new Map(overridesRes.data.map((o) => [o.date, o.capacity]));
  const dates = new Set([...Object.keys(occupancy), ...overrides.keys()]);
  for (const date of dates) {
    const limit = overrides.get(date) ?? settings.daily_capacity;
    result[date] = Math.max(limit - (occupancy[date] ?? 0), 0);
  }
  return result;
}
```

```ts
  reservationId: string; code: string; email: string;
  firstName: string; lastName: string; idType: string; idNumber: string;
  date: string; adults: number; children: number;
  totalUsd: number; totalVes: number; bcvRate: number;
}

export interface BookingDeps {
  getPrices: () => Promise<{ adult: number; child: number }>;
  reserve: (input: {
    date: string; firstName: string; lastName: string; email: string;
    idType: 'V' | 'E' | 'P' | 'J'; idNumber: string;
    adults: number; children: number; totalUsd: number;

  getRate: () => Promise<number | null>;
  pay: (req: C2pRequest) => Promise<C2pResult>;
  confirm: (id: string, p: { bcvRate: number; totalVes: number; paymentRef: string }) => Promise<void>;
  fail: (id: string) => Promise<void>;
  sendVoucher: (v: VoucherData) => Promise<void>;

export type BookingResult =
  | { ok: true; reservation: { id: string; code: string; totalUsd: number; totalVes: number; bcvRate: number } }
  | { ok: false; error: 'validation'; message: string }
  | { ok: false; error: 'capacity'; remaining: number }
  | { ok: false; error: 'rate_unavailable'; message: string }
  | { ok: false; error: 'payment_declined'; message: string }
  | { ok: false; error: 'payment_pending'; message: string };

export async function processBooking(raw: unknown, deps: BookingDeps): Promise<BookingResult> {
  const parsed = paymentSchema.safeParse(raw);
  if (!parsed.success) {
    return { ok: false, error: 'validation', message: parsed.error.issues[0]?.message ?? 'Datos inválidos' };
  }
  const input = parsed.data;

  const prices = await deps.getPrices();
  const usd = totalUsd(input.adults, input.children, prices);

  const reserved = await deps.reserve({
    date: input.date, firstName: input.firstName, lastName: input.lastName,
    email: input.email, idType: input.idType, idNumber: input.idNumber,
    adults: input.adults, children: input.children, totalUsd: usd,
  });
  if (!reserved.ok) return { ok: false, error: 'capacity', remaining: reserved.remaining };

  const rate = await deps.getRate();
  if (rate === null) {
    await deps.fail(reserved.id);
    return {
      ok: false, error: 'rate_unavailable',
      message: 'No pudimos obtener la tasa BCV. Intenta de nuevo más tarde.',
    };
  }
  const ves = usdToVes(usd, rate);

  const payment = await deps.pay({
    amountVes: ves,
    destinationBankCode: input.bankCode,
    destinationId: `${input.idType}${input.idNumber}`,
    destinationPhone: input.phone,
    otp: input.otp,
    invoiceNumber: reserved.code,
  });

  if (payment.status === 'declined') {
    await deps.fail(reserved.id);
    return { ok: false, error: 'payment_declined', message: payment.message };
  }
  if (payment.status === 'unknown') {
    // Ambiguous: may have charged. Keep pending; admin verifies manually. Never auto-retry.
    return {
      ok: false, error: 'payment_pending',
      message: 'Tu pago está siendo verificado. Te contactaremos con la confirmación.',
    };
  }

  await deps.confirm(reserved.id, { bcvRate: rate, totalVes: ves, paymentRef: payment.reference });

  try {
    await deps.sendVoucher({
      reservationId: reserved.id, code: reserved.code, email: input.email,
      firstName: input.firstName, lastName: input.lastName,
      idType: input.idType, idNumber: input.idNumber,
      date: input.date, adults: input.adults, children: input.children,
      totalUsd: usd, totalVes: ves, bcvRate: rate,
    });
  } catch (e) {
    console.error('voucher email failed (booking still confirmed):', e);
  }

  return { ok: true, reservation: { id: reserved.id, code: reserved.code, totalUsd: usd, totalVes: ves, bcvRate: rate } };
}
```

```ts
import QRCode from 'qrcode';

const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export async function GET(
  _req: Request,
  ctx: { params: Promise<{ id: string }> }
) {
  const { id } = await ctx.params;
  if (!UUID_RE.test(id)) return new Response('Bad request', { status: 400 });
  const url = `${process.env.NEXT_PUBLIC_APP_URL}/admin/checkin/${id}`;
  const png = await QRCode.toBuffer(url, { width: 360, margin: 1 });
  return new Response(new Uint8Array(png), {
    status: 200,
    headers: { 'Content-Type': 'image/png', 'Cache-Control': 'public, max-age=31536000, immutable' },
  });
}
```

```tsx
'use client';

import { useState } from 'react';
import { z } from 'zod';
import { paymentSchema, type BookingInput } from '@/lib/schemas';
import { VENEZUELAN_BANKS } from '@/lib/banks';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { Card, CardContent } from '@/components/ui/card';
import {
  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
} from '@/components/ui/select';

const payFieldsSchema = paymentSchema.pick({
  idType: true, idNumber: true, phone: true, bankCode: true, otp: true,
});
export type PayFields = z.infer<typeof payFieldsSchema>;

export function StepPayment(props: {
  booking: BookingInput;
  totalUsd: number;
  vesInfo?: { rate: number; totalVes: number } | null;
  onSubmit: (fields: PayFields) => void;
  submitting: boolean;
  error: string | null;
}) {
  const [fields, setFields] = useState<PayFields>({
    idType: 'V', idNumber: '', phone: '', bankCode: '', otp: '',
  });
  const [errors, setErrors] = useState<Record<string, string>>({});

  function submit() {
    const parsed = payFieldsSchema.safeParse(fields);
    if (!parsed.success) {
      const errs: Record<string, string> = {};
      for (const issue of parsed.error.issues) errs[String(issue.path[0])] = issue.message;
      setErrors(errs);
      return;
    }
    props.onSubmit(parsed.data);
  }

  return (
    <div className="space-y-4">
      <div className="rounded-lg bg-muted p-4">
        <div className="flex justify-between">
          <span className="text-sm text-muted-foreground">Total a pagar</span>
          <span data-testid="pay-total-usd" className="text-xl font-bold">
            ${props.totalUsd.toFixed(2)}
          </span>
        </div>
        {props.vesInfo && (
          <p className="mt-1 text-right text-sm text-muted-foreground">
            Bs. {props.vesInfo.totalVes.toFixed(2)} (tasa BCV {props.vesInfo.rate})
          </p>
        )}
      </div>

      <Card>
        <CardContent className="space-y-3 pt-4">
          <div className="flex gap-2">
            <div className="w-24">
              <Label htmlFor="idType">Tipo</Label>
              <Select value={fields.idType}
                onValueChange={(v) => setFields({ ...fields, idType: v as PayFields['idType'] })}>

                  {(['V', 'E', 'P', 'J'] as const).map((t) => (
                    <SelectItem key={t} value={t}>{t}</SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>

              <Label htmlFor="idNumber">Número de documento</Label>

          </div>

            <Label htmlFor="phone">Teléfono (Pago Móvil)</Label>
```

```tsx
'use client';

import { useState } from 'react';
import { format } from 'date-fns';
import { StepDetails } from './step-details';
import { StepPayment, type PayFields } from './step-payment';
import { Confirmation } from './confirmation';
import { getAvailability, submitBookingAndPay } from '@/app/actions/booking';
import { totalUsd } from '@/lib/money';
import type { BookingInput } from '@/lib/schemas';
import type { BookingResult } from '@/lib/booking/process';

type Confirmed = Extract<BookingResult, { ok: true }>['reservation'];

export function BookingWizard(props: {
  prices: { adult: number; child: number };
  initialAvailability: Record<string, number>;
}) {
  const [step, setStep] = useState<'details' | 'payment' | 'done'>('details');
  const [availability, setAvailability] = useState(props.initialAvailability);
  const [booking, setBooking] = useState<BookingInput>({
    date: format(new Date(), 'yyyy-MM-dd'),
    adults: 1, children: 0, firstName: '', lastName: '', email: '',
  });
  const [idLabel, setIdLabel] = useState('');
  const [reservation, setReservation] = useState<Confirmed | null>(null);
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handlePay(fields: PayFields) {
    setSubmitting(true);
    setError(null);
    const result = await submitBookingAndPay({ ...booking, ...fields });
    setSubmitting(false);
    if (result.ok) {
      setIdLabel(`${fields.idType}-${fields.idNumber}`);
      setReservation(result.reservation);
      setStep('done');
      return;
    }
    switch (result.error) {
      case 'capacity':
        setError(`Ya no hay cupo suficiente para esa fecha. Quedan ${result.remaining} puestos.`);
        setStep('details');
        break;
      case 'payment_declined':
        setError(`Pago rechazado: ${result.message}. Genera una nueva clave C2P e intenta de nuevo.`);
        break;
      default:
        setError(result.message);
    }
  }

  if (step === 'done' && reservation) {
    return <Confirmation booking={booking} reservation={reservation} idLabel={idLabel} />;
  }
  if (step === 'payment') {
    return (
      <StepPayment
        booking={booking}
        totalUsd={totalUsd(booking.adults, booking.children, props.prices)}
        onSubmit={handlePay}
        submitting={submitting}
        error={error}
      />
    );
  }
  return (
    <>
      {error && <p role="alert" className="mb-3 rounded-md bg-destructive/10 p-3 text-sm text-destructive">{error}</p>}
      <StepDetails
        prices={props.prices}
        availability={availability}
        initial={booking}
        onMonthChange={async (m) => setAvailability(await getAvailability(m))}
        onNext={(data) => { setBooking(data); setError(null); setStep('payment'); }}
      />
    </>
  );
}
```

```ts
import { createServerClient } from '@supabase/ssr';
import { NextResponse, type NextRequest } from 'next/server';

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({ request });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => request.cookies.getAll(),
        setAll: (cookiesToSet) => {
          cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value));
          response = NextResponse.next({ request });
          cookiesToSet.forEach(({ name, value, options }) =>
            response.cookies.set(name, value, options)
          );
        },
      },
    }
  );

  const { data: { user } } = await supabase.auth.getUser();

  const isAdminPath = request.nextUrl.pathname.startsWith('/admin');
  const isLogin = request.nextUrl.pathname === '/admin/login';

  if (isAdminPath && !isLogin && !user) {
    const url = request.nextUrl.clone();
    url.pathname = '/admin/login';
    url.searchParams.set('next', request.nextUrl.pathname);
    return NextResponse.redirect(url);
  }
  return response;
}

export const config = { matcher: ['/admin/:path*'] };
```

```tsx
import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { Card, CardContent } from '@/components/ui/card';
import type { ReservationRow } from '@/lib/db/reservations';

type BannerKind = 'valid' | 'wrong-date' | 'already-in' | 'invalid';

function classify(r: ReservationRow, today: string): BannerKind {
  if (r.status === 'checked_in') return 'already-in';
  if (r.status !== 'confirmed') return 'invalid';
  if (r.date !== today) return 'wrong-date';
  return 'valid';
}

const BANNER: Record<BannerKind, { text: (r: ReservationRow) => string; cls: string }> = {
  valid: { text: () => '✅ Reserva válida para hoy', cls: 'bg-green-100 text-green-900' },
  'wrong-date': {
    text: (r) => `⚠️ Fecha distinta: la reserva es para ${r.date}`,
    cls: 'bg-yellow-100 text-yellow-900',
  },
  'already-in': {
    text: (r) => `⛔ Ya registró entrada (${r.checked_in_at?.slice(11, 16) ?? ''})`,
    cls: 'bg-red-100 text-red-900',
  },
  invalid: { text: (r) => `⛔ Reserva no válida (estado: ${r.status})`, cls: 'bg-red-100 text-red-900' },
};

  reservation: ReservationRow;
  today: string;
  onCheckIn: (id: string) => Promise<void> | void;
}) {
```

```ts
import { test, expect } from '@playwright/test';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

// A weekday ~3 weeks out, always in the future relative to "today"
function targetDate(): string {
  const d = new Date();
  d.setDate(d.getDate() + 21);
  return d.toISOString().slice(0, 10);
}

async function fillBookingStep(page: import('@playwright/test').Page, date: string) {
  const day = Number(date.slice(8, 10));
  // navigate the calendar to the target month if needed, then click the day
  const target = new Date(`${date}T12:00:00`);
  const monthsAhead =
    (target.getFullYear() - new Date().getFullYear()) * 12 +
    (target.getMonth() - new Date().getMonth());
  for (let i = 0; i < monthsAhead; i++) {
    await page.getByRole('button', { name: /next month|mes siguiente/i }).click();
  }
  await page.getByRole('gridcell', { name: String(day), exact: true }).first().click();
  await page.getByRole('button', { name: /aumentar adultos/i }).click(); // 2 adults
  await page.getByLabel(/nombre/i).fill('Ana');
  await page.getByLabel(/apellido/i).fill('Pérez');
  await page.getByLabel(/correo/i).fill('ana@example.com');
  await page.getByRole('button', { name: /continuar/i }).click();
}

async function fillPaymentStep(page: import('@playwright/test').Page, otp: string) {
  await page.getByLabel(/número de documento/i).fill('12345678');
  await page.getByLabel(/teléfono/i).fill('04141234567');
  await page.getByLabel(/^banco$/i).click();
  await page.getByRole('option', { name: /banesco/i }).click();
  await page.getByLabel(/clave de pago/i).fill(otp);
  await page.getByRole('button', { name: /pagar/i }).click();
}

test.describe('booking flow', () => {
  const DATE = targetDate();

  test.beforeEach(async () => {
    await supabase.from('reservations').delete().eq('date', DATE);
    await supabase.from('capacity_overrides').delete().eq('date', DATE);
  });

  test('happy path: books, pays (mock), sees confirmation with code', async ({ page }) => {
    await page.goto('/');
    await fillBookingStep(page, DATE);
    await fillPaymentStep(page, '123456');
    await expect(page.getByText(/reserva confirmada/i)).toBeVisible({ timeout: 20_000 });
    const code = await page.getByTestId('reservation-code').textContent();
    expect(code).toMatch(/^PP-/);

    const { data } = await supabase
      .from('reservations').select('status').eq('code', code!).single();
    expect(data!.status).toBe('confirmed');
  });

  test('declined payment: shows error, reservation released', async ({ page }) => {
    await page.goto('/');
    await fillBookingStep(page, DATE);
    await fillPaymentStep(page, '999999'); // mock decline
    await expect(page.getByText(/pago rechazado/i)).toBeVisible({ timeout: 20_000 });

    const { data } = await supabase
      .from('reservations').select('status').eq('date', DATE);
    expect(data!.every((r) => r.status === 'failed')).toBe(true);
  });

  test('sold out: capacity error reports remaining slots', async ({ page }) => {
    await supabase.from('capacity_overrides').insert({ date: DATE, capacity: 1 });
    await page.goto('/');
    await fillBookingStep(page, DATE); // requests 2 adults > capacity 1
    await fillPaymentStep(page, '123456');
    await expect(page.getByText(/no hay cupo suficiente/i)).toBeVisible({ timeout: 20_000 });
  });
});
```
